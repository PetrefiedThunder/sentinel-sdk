    17	@dataclass
    18	class PageResult:
    19	    """One page of results from a cursor-paginated list endpoint.
    20	
    21	    Mirrors the JS SDK's PageResult: `data` is the list of items, `has_more`
    22	    indicates whether another page exists, and `next_cursor` is the opaque
    23	    cursor to pass back to fetch it (None when there are no more pages).
    24	    """
    25	
    26	    data: list = field(default_factory=list)
    27	    has_more: bool = False
    28	    next_cursor: str | None = None
    29	
    30	    @classmethod
    31	    def from_envelope(cls, body: Any) -> PageResult:
    32	        """Build a PageResult from the API's {data, has_more, next_cursor} body.
    33	
    34	        Tolerates a bare list (older servers that return an unwrapped array):
    35	        the whole list becomes a single terminal page.
    36	        """
    37	        if isinstance(body, dict):
    38	            return cls(
    39	                data=body.get("data") or [],
    40	                has_more=bool(body.get("has_more", False)),
    41	                next_cursor=body.get("next_cursor"),
    42	            )
    43	        return cls(data=list(body or []), has_more=False, next_cursor=None)

    46	def _ensure_json_serializable(arguments: Any) -> None:
    47	    """Fail fast if `arguments` can't be JSON-encoded.
    48	
    49	    Without this check the POST silently succeeds (httpx converts the value
    50	    via its own serializer, or the server stores something useless), and
    51	    the agent then hangs until `timeout_seconds` waiting on a decision it
    52	    can never get. Raise immediately with the offending type so the caller
    53	    sees the real error.
    54	    """
    55	    try:
    56	        json.dumps(arguments)
    57	    except TypeError as e:
    58	        raise TypeError(
    59	            f"@oversight arguments must be JSON-serializable. "
    60	            f"Got an un-serializable value: {e}. "
    61	            f"Convert sets/objects/bytes to plain dict/list/str/int/float/bool/None before the call."
    62	        ) from e

    65	def _raise_for_status(r: httpx.Response) -> None:
    66	    """Translate non-2xx into SentinelAPIError with the API's error body."""
    67	    if r.is_success:
    68	        return
    69	    detail = ""
    70	    try:
    71	        body = r.json()
    72	        if isinstance(body, dict):
    73	            detail = body.get("detail") or body.get("message") or str(body)
    74	        else:
    75	            detail = str(body)
    76	    except Exception:
    77	        detail = (r.text or "")[:500]
    78	    raise SentinelAPIError(r.status_code, detail, url=str(r.request.url))

    83

   106	    def _get_client(self) -> httpx.Client:
   107	        if self._client is None:
   108	            self._client = httpx.Client(
   109	                base_url=self.config.api_url.rstrip("/"),
   110	                headers=self._headers(),
   111	                timeout=60.0,
   112	                limits=httpx.Limits(max_keepalive_connections=10, max_connections=20),
   113	            )

   117	        if self._aclient is None:
   118	            self._aclient = httpx.AsyncClient(
   119	                base_url=self.config.api_url.rstrip("/"),
   120	                headers=self._headers(),
   121	                timeout=60.0,
   122	                limits=httpx.Limits(max_keepalive_connections=10, max_connections=20),
   123	            )
   124	        return self._aclient

   126	    def close(self) -> None:
   127	        if self._client is not None:

   131	    async def aclose(self) -> None:

   189	    def wait_for_decision(
   190	        self,
   191	        action_id: str,
   192	        timeout: float | None = None,
   193	        poll_interval: float | None = None,
   194	    ) -> dict:
   195	        """Block until the approval is decided or the timeout elapses.
   196	
   197	        Uses the server-side long-poll endpoint when available (single RTT per
   198	        ~30s wait window), falling back to client polling.
   199	        """
   200	        timeout = timeout or self.config.timeout_seconds
   201	        deadline = time.monotonic() + timeout
   202	        while True:
   203	            remaining = max(1.0, min(30.0, deadline - time.monotonic()))
   204	            try:
   205	                r = self._get_client().get(
   206	                    f"/v1/approvals/{action_id}/wait",
   207	                    params={"timeout": remaining},
   208	                    timeout=remaining + 5.0,
   209	                )
   210	                _raise_for_status(r)
   211	                data = r.json()
   212	            except SentinelAPIError as e:
   213	                # Fall back to plain polling if /wait isn't available (older server)
   214	                if e.status_code != 404:
   215	                    raise
   216	                data = self.get_approval(action_id)
   217	            status = data.get("status") or data.get("decision")
   218	            if status in ("approved", "rejected"):
   219	                return data
   220	            if time.monotonic() >= deadline:
   221	                raise ApprovalTimeout(action_id=action_id, timeout_seconds=timeout)
