2: 
3: CrewAI tools are typically built with @tool() or BaseTool. Wrap them with
4: `gated()` to require human approval before each invocation:
5: 
6:     from crewai import Agent, Task, Crew
7:     from crewai.tools import tool
8:     from sentinel.adapters.crewai import gated
9: 
10:     @tool("wire_transfer")
11:     def _wire_transfer(amount: int, recipient: str) -> dict:
12:         return stripe.transfers.create(amount=amount, destination=recipient)
13: 
14:     wire_transfer = gated(
15:         _wire_transfer,
16:         risk_level="high",
17:         approvers=["ops@acme.com"],
18:     )
19: 
20:     agent = Agent(tools=[wire_transfer], ...)
21: 
22: The adapter wraps the underlying callable so CrewAI calls it normally —
23: Sentinel intercepts inside the wrapper and pauses for human approval.
24: """
25: 
26: from __future__ import annotations
27: 
28: from typing import TYPE_CHECKING, Any
29: 
30: from ..client import SentinelClient
31: from ..exceptions import ApprovalRejected
32: 
33: if TYPE_CHECKING:
34:     from collections.abc import Callable
35: 
36: 
37: def gated(
38:     tool: Any,
39:     *,
40:     client: SentinelClient | None = None,
41:     risk_level: str = "high",
42:     approvers: list[str] | None = None,
43:     timeout_seconds: float | None = None,
44:     function_name: str | None = None,
45: ) -> Any:
46:     """Wrap a CrewAI tool (BaseTool instance or @tool-decorated callable)
47:     so each invocation pauses for Sentinel approval.
48: 
49:     Returns an object with the same external interface — CrewAI can call
50:     it exactly like the original tool.
51:     """
52:     sentinel_client = client or SentinelClient()
53:     derived_name = function_name or _derive_tool_name(tool)
54: 
55:     original = _extract_callable(tool)
56: 
57:     def wrapper(*args: Any, **kwargs: Any) -> Any:
58:         # CrewAI tools are usually called with kwargs; fall back to positional.
59:         arguments = kwargs if kwargs else {"args": list(args)}
60:         approval = sentinel_client.create_approval(
61:             function_name=derived_name,
62:             arguments=arguments,
63:             risk_level=risk_level,
64:             approvers=approvers,
65:             timeout_seconds=timeout_seconds,
66:         )
67:         action_id = approval.get("action_id") or approval.get("id")
68:         decision = sentinel_client.wait_for_decision(action_id, timeout=timeout_seconds)
69:         status = decision.get("status") or decision.get("decision")
70:         if status != "approved":
71:             raise ApprovalRejected(
72:                 reason=decision.get("reason", "Tool execution not approved"),
73:                 action_id=action_id,
74:             )
75:         return original(*args, **kwargs)
76: 
77:     # Preserve the CrewAI BaseTool object if we were handed one — only
78:     # swap out the underlying _run / func reference.
79:     if hasattr(tool, "_run") and callable(tool._run):
80:         tool._run = wrapper  # type: ignore[attr-defined]
81:         return tool
82:     if hasattr(tool, "func") and callable(tool.func):
83:         tool.func = wrapper  # type: ignore[attr-defined]
84:         return tool
85:     # Plain callable — return the wrapper directly
86:     wrapper.__name__ = getattr(original, "__name__", derived_name)
87:     return wrapper
88: 
89: 
90: def _extract_callable(tool: Any) -> Callable:
91:     if hasattr(tool, "_run") and callable(tool._run):
92:         return tool._run
93:     if hasattr(tool, "func") and callable(tool.func):
94:         return tool.func
95:     if callable(tool):
96:         return tool
97:     raise TypeError(
98:         f"Don't know how to extract a callable from {type(tool).__name__}. "
99:         "Pass a CrewAI BaseTool instance or a @tool-decorated function."
100:     )
101: 
102: 
103: def _derive_tool_name(tool: Any) -> str:
104:     for attr in ("name", "tool_name", "__name__"):
105:         value = getattr(tool, attr, None)
106:         if isinstance(value, str) and value:
107:             return value
108:     return "crewai_tool"

1: """Semantic Kernel adapter — gate kernel function/tool invocation behind
2: Sentinel approval.
3: 
4: Semantic Kernel (the `semantic-kernel` package) runs every `KernelFunction`
5: through a *function-invocation filter* pipeline. A filter is an async callable
6: ``(context, next)`` where calling ``await next(context)`` executes the function;
7: not calling it skips execution. That makes filters the natural place to pause
8: for human approval before any plugin/@kernel_function runs.
9: 
10:     from semantic_kernel import Kernel
11:     from sentinel.adapters.semantic_kernel import gate_kernel
12: 
13:     kernel = Kernel()
14:     kernel.add_plugin(OpsPlugin(), plugin_name="ops")
15: 
16:     gate_kernel(kernel, risk_level="high", approvers=["ops@acme.com"])
17:     # Every function the kernel invokes now pauses for Sentinel approval.
18: 
19: Use ``function_allowlist`` to gate only specific functions, or
20: ``function_denylist`` to exempt specific ones; ungated functions call through
21: directly with no approval round-trip. Names are matched against both the bare
22: function name (e.g. ``wire_transfer``) and the qualified ``plugin.function``
23: form (e.g. ``ops.wire_transfer``).
24: 
25: If you build the filter yourself, ``sentinel_filter()`` returns the raw filter
26: callable to register via ``kernel.add_filter(FilterTypes.FUNCTION_INVOCATION, ...)``
27: or the ``@kernel.filter`` decorator.
28: 
29: This module never imports `semantic-kernel`, so it works without the package
30: installed and across versions.
31: """

34: 
35: from typing import TYPE_CHECKING, Any
36: 
37: if TYPE_CHECKING:
38:     from collections.abc import Awaitable, Callable, Iterable
39: 
40: from ..client import SentinelClient
41: from ..exceptions import ApprovalRejected
42: 
43: 
44: def gate_kernel(
45:     kernel: Any,
46:     *,
47:     client: SentinelClient | None = None,

49:     approvers: list[str] | None = None,
50:     timeout_seconds: float | None = None,
51:     function_allowlist: Iterable[str] | None = None,
52:     function_denylist: Iterable[str] | None = None,
53: ) -> Any:
54:     """Register a Sentinel function-invocation filter on a Semantic Kernel
55:     ``Kernel`` so each function invocation pauses for human approval.
56: 
57:     Returns the same ``kernel`` object so calls can be chained. The kernel is
58:     passed in untyped — this adapter never imports ``semantic-kernel``.
59:     """
60:     sk_filter = sentinel_filter(
61:         client=client,
62:         risk_level=risk_level,
63:         approvers=approvers,
64:         timeout_seconds=timeout_seconds,
65:         function_allowlist=function_allowlist,
66:         function_denylist=function_denylist,
67:     )
68:     # `FilterTypes.FUNCTION_INVOCATION` == "function_invocation"; add_filter
69:     # accepts the enum or its string value, so pass the string to avoid
70:     # importing semantic-kernel here.
71:     kernel.add_filter("function_invocation", sk_filter)
72:     return kernel
73: 
74: 
75: def sentinel_filter(
76:     *,
77:     client: SentinelClient | None = None,
78:     risk_level: str = "high",
79:     approvers: list[str] | None = None,
80:     timeout_seconds: float | None = None,
81:     function_allowlist: Iterable[str] | None = None,
82:     function_denylist: Iterable[str] | None = None,
83: ) -> Callable[[Any, Callable[[Any], Awaitable[None]]], Awaitable[None]]:
84:     """Build a Semantic Kernel function-invocation filter that requires
85:     Sentinel approval before each gated function runs.
86: 
87:     The returned async callable matches Semantic Kernel's filter signature
88:     ``async def filter(context, next)`` and can be registered with
89:     ``kernel.add_filter(FilterTypes.FUNCTION_INVOCATION, ...)`` or the
90:     ``@kernel.filter(FilterTypes.FUNCTION_INVOCATION)`` decorator.
91:     """
92:     sentinel_client = client or SentinelClient()
93:     allow = set(function_allowlist) if function_allowlist is not None else None
94:     deny = set(function_denylist) if function_denylist is not None else None
95: 
96:     def _should_gate(name: str, qualified_name: str) -> bool:
97:         if allow is not None:
98:             return name in allow or qualified_name in allow
99:         if deny is not None:
100:             return name not in deny and qualified_name not in deny
101:         return True
102: 
103:     async def filter(context: Any, next: Callable[[Any], Awaitable[None]]) -> None:
104:         name = _function_name(context)
105:         qualified_name = _qualified_function_name(context)
106:         if not _should_gate(name, qualified_name):
107:             await next(context)
108:             return
109: 
110:         approval = await sentinel_client.acreate_approval(
111:             function_name=qualified_name,
112:             arguments=_arguments(context),
113:             risk_level=risk_level,
114:             approvers=approvers,
115:             timeout_seconds=timeout_seconds,
116:         )
117:         action_id = approval.get("action_id") or approval.get("id")
118:         decision = await sentinel_client.await_for_decision(action_id, timeout=timeout_seconds)
119:         status = decision.get("status") or decision.get("decision")
120:         if status != "approved":
121:             raise ApprovalRejected(
122:                 reason=decision.get("reason", "Function execution not approved"),
123:                 action_id=action_id,
124:             )
125:         await next(context)
126: 
127:     return filter
128: 
129: 
130: def _function_name(context: Any) -> str:
131:     function = getattr(context, "function", None)
132:     name = getattr(function, "name", None)
133:     if isinstance(name, str) and name:
134:         return name
135:     return "semantic_kernel_function"
136: 
137: 
138: def _qualified_function_name(context: Any) -> str:
139:     function = getattr(context, "function", None)
140:     name = _function_name(context)
141:     plugin = getattr(function, "plugin_name", None)
142:     if isinstance(plugin, str) and plugin:
143:         return f"{plugin}.{name}"
144:     return name
145: 
146: 
147: def _arguments(context: Any) -> dict[str, Any]:
148:     """KernelArguments behaves like a mapping; coerce it to a plain JSON-safe
149:     dict, dropping the non-serializable execution-settings entry SK adds."""
150:     arguments = getattr(context, "arguments", None)
151:     if arguments is None:
152:         return {}
153:     try:
154:         items = dict(arguments)
155:     except (TypeError, ValueError):
156:         return {"args": str(arguments)}
157:     return {k: v for k, v in items.items() if k != "execution_settings"}
