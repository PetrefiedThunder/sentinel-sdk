1: """OpenAI Agents SDK adapter — wrap an Agents SDK tool with Sentinel approval.
2: 
3: Tools in the OpenAI Agents SDK (`openai-agents` package) are typically built
4: with @function_tool, which produces a FunctionTool whose async
5: `on_invoke_tool(ctx, args_json)` callback executes the tool. Wrap a tool with
6: `gated()` to require human approval before each invocation:
7: 
8:     from agents import Agent, Runner, function_tool
9:     from sentinel.adapters.openai_agents import gated
10: 
11:     @function_tool
12:     def wire_transfer(amount: int, recipient: str) -> dict:
13:         return stripe.transfers.create(amount=amount, destination=recipient)
14: 
15:     agent = Agent(
16:         name="ops",
17:         tools=[gated(wire_transfer, risk_level="high", approvers=["ops@acme.com"])],
18:     )
19:     result = Runner.run_sync(agent, "...")
20: 
21: The adapter swaps the tool's `on_invoke_tool` callback for one that pauses
22: for Sentinel approval, so the Agents SDK calls the tool normally. Plain
23: callables (functions not yet wrapped by @function_tool) are also accepted —
24: they get the same gating applied directly. This module never imports
25: `openai-agents`, so it works without the package installed and across
26: versions.
27: """
28: 
29: from __future__ import annotations
30: 
31: import asyncio
32: import functools
33: import json
34: from typing import TYPE_CHECKING, Any
35: 
36: if TYPE_CHECKING:
37:     from collections.abc import Callable
38: 
39: from ..client import SentinelClient
40: from ..exceptions import ApprovalRejected
41: 
42: 
43: def gated(
44:     tool: Any,
45:     *,
46:     client: SentinelClient | None = None,
47:     risk_level: str = "high",
48:     approvers: list[str] | None = None,
49:     timeout_seconds: float | None = None,
50:     function_name: str | None = None,
51: ) -> Any:
52:     """Wrap an OpenAI Agents SDK tool (FunctionTool instance or plain
53:     callable) so each invocation pauses for Sentinel approval.
54: 
55:     Returns the same FunctionTool object with its `on_invoke_tool` callback
56:     gated, or a gated wrapper if a plain callable was passed.
57:     """
58:     sentinel_client = client or SentinelClient()
59:     derived_name = function_name or _derive_tool_name(tool)
60: 
61:     if hasattr(tool, "on_invoke_tool") and callable(tool.on_invoke_tool):
62:         original = tool.on_invoke_tool
63: 
64:         @functools.wraps(original)
65:         async def gated_invoke(ctx: Any, args_json: str) -> Any:
66:             arguments = _parse_arguments(args_json)
67:             await _arequire_approval(
68:                 sentinel_client,
69:                 derived_name,
70:                 arguments,
71:                 risk_level=risk_level,
72:                 approvers=approvers,
73:                 timeout_seconds=timeout_seconds,
74:             )
75:             return await original(ctx, args_json)
76: 
77:         tool.on_invoke_tool = gated_invoke
78:         return tool
79: 
80:     if callable(tool):
81:         return _wrap_plain_callable(
82:             tool,
83:             sentinel_client,
84:             derived_name,
85:             risk_level=risk_level,
86:             approvers=approvers,
87:             timeout_seconds=timeout_seconds,
88:         )
89: 
90:     raise TypeError(
91:         f"Don't know how to gate {type(tool).__name__}. "
92:         "Pass a FunctionTool (from @function_tool) or a plain callable."
93:     )

101:     risk_level: str,

113:                 arguments,
114:                 risk_level=risk_level,

116:                 timeout_seconds=timeout_seconds,
117:             )
118:             return await fn(*args, **kwargs)
119: 
120:         return async_wrapper

128:             risk_level=risk_level,

136:             raise ApprovalRejected(
137:                 reason=decision.get("reason", "Tool execution not approved"),

140:         return fn(*args, **kwargs)

156:         arguments=arguments,

171: def _parse_arguments(args_json: Any) -> dict[str, Any]:
172:     """on_invoke_tool receives the tool arguments as a JSON string."""
173:     if isinstance(args_json, dict):
174:         return args_json
175:     if not args_json:
176:         return {}
177:     try:
178:         parsed = json.loads(args_json)
179:     except (TypeError, ValueError):
180:         return {"raw": str(args_json)}
181:     if isinstance(parsed, dict):
182:         return parsed
183:     return {"args": parsed}

188:         value = getattr(tool, attr, None)
189:         if isinstance(value, str) and value:
190:             return value
191:     return "openai_agents_tool"

2: Sentinel approval.
3: 
4: `SentinelToolGate` wraps plain Python callables (the functions you pass to
